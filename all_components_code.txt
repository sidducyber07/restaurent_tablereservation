import { createSlice } from "@reduxjs/toolkit";
import { STORAGE_KEY } from "./middleware.js";

const loadStoredReservations = () => {
  try {
    const raw = localStorage.getItem(STORAGE_KEY);
    return raw ? JSON.parse(raw) : [];
  } catch (err) {
    console.error("Failed to load reservations from localStorage:", err);
    return [];
  }
};

const initialState = {
  selectedRestaurant: null,
  draft: {
    date: new Date().toISOString().split("T")[0],
    timeSlot: "",
    guests: 2,
    selectedTable: null,
    customerName: "",
    customerEmail: "",
    customerPhone: "",
    specialRequests: ""
  },
  confirmedReservations: loadStoredReservations(),
  actionLogs: [],
  isLoading: false,
  error: null
};

export const reservationSlice = createSlice({
  name: "reservation",
  initialState,
  reducers: {
    setSelectedRestaurant: (state, action) => {
      state.selectedRestaurant = action.payload;
    },

    updateDraft: (state, action) => {
      state.draft = { ...state.draft, ...action.payload };
    },

    resetDraft: (state) => {
      state.draft = {
        date: new Date().toISOString().split("T")[0],
        timeSlot: "",
        guests: 2,
        selectedTable: null,
        customerName: "",
        customerEmail: "",
        customerPhone: "",
        specialRequests: ""
      };
      state.error = null;
    },

    // Set full list from GET /api/reservations
    setConfirmedReservations: (state, action) => {
      state.confirmedReservations = action.payload;
      state.isLoading = false;
      state.error = null;
    },

    // POST /api/reservations
    confirmReservation: (state, action) => {
      state.confirmedReservations.unshift(action.payload);
      state.isLoading = false;
      state.error = null;
    },

    // PUT /api/reservations/:id
    updateReservationInState: (state, action) => {
      const index = state.confirmedReservations.findIndex(r => r.id === action.payload.id);
      if (index !== -1) {
        state.confirmedReservations[index] = { ...state.confirmedReservations[index], ...action.payload };
      }
      state.isLoading = false;
    },

    // DELETE /api/reservations/:id
    cancelReservation: (state, action) => {
      state.confirmedReservations = state.confirmedReservations.filter(
        (r) => r.id !== action.payload
      );
    },

    clearAllReservations: (state) => {
      state.confirmedReservations = [];
    },

    addLogEntry: (state, action) => {
      state.actionLogs.unshift(action.payload);
      if (state.actionLogs.length > 25) {
        state.actionLogs.pop();
      }
    },

    setLoading: (state, action) => {
      state.isLoading = action.payload;
    },

    setError: (state, action) => {
      state.error = action.payload;
      state.isLoading = false;
    }
  }
});

export const {
  setSelectedRestaurant,
  updateDraft,
  resetDraft,
  setConfirmedReservations,
  confirmReservation,
  updateReservationInState,
  cancelReservation,
  clearAllReservations,
  addLogEntry,
  setLoading,
  setError
} = reservationSlice.actions;

export default reservationSlice.reducer;

/**
 * Custom Redux Middleware:
 * 1. reservationActionMonitorMiddleware: Monitors & audits reservation actions in real-time
 * 2. storageSyncMiddleware: Synchronizes reservations state with localStorage
 */

export const STORAGE_KEY = "gourmet_reservations";

export const reservationActionMonitorMiddleware = (store) => (next) => (action) => {
  if (typeof action?.type === "string" && action.type.startsWith("reservation/")) {
    const timestamp = new Date().toLocaleTimeString();
    console.log(
      `%c[Redux Middleware @ ${timestamp}] %c${action.type}`,
      "color: #ea580c; font-weight: bold;",
      "color: #2563eb; font-weight: 600;",
      action.payload !== undefined ? action.payload : ""
    );

    // Prevent recursive loop if action is the logger action itself
    if (action.type !== "reservation/addLogEntry") {
      setTimeout(() => {
        store.dispatch({
          type: "reservation/addLogEntry",
          payload: {
            id: `log_${Date.now()}_${Math.random().toString(36).substr(2, 4)}`,
            type: action.type,
            time: timestamp,
            payloadSummary: action.payload ? JSON.stringify(action.payload).slice(0, 100) : "none"
          }
        });
      }, 0);
    }
  }

  const result = next(action);
  return result;
};

export const storageSyncMiddleware = (store) => (next) => (action) => {
  const result = next(action);

  const syncActions = [
    "reservation/setConfirmedReservations",
    "reservation/confirmReservation",
    "reservation/updateReservationInState",
    "reservation/cancelReservation",
    "reservation/clearAllReservations"
  ];

  if (syncActions.includes(action.type)) {
    try {
      const state = store.getState();
      const reservations = state.reservation?.confirmedReservations || [];
      localStorage.setItem(STORAGE_KEY, JSON.stringify(reservations));
      console.log(`%c[Storage Sync Middleware] Synchronized ${reservations.length} bookings to localStorage`, "color: #16a34a; font-weight: bold;");
    } catch (err) {
      console.error("[Storage Sync Middleware] Failed to write to localStorage:", err);
    }
  }

  return result;
};

/**
 * REST API Client implementation using standard window.fetch()
 * Supports Full CRUD: GET, POST, PUT/PATCH, and DELETE
 */
const BASE = "/api";

export const fetchClient = {
  async request(url, { method = "GET", params, body, signal, simulateError, simulateDelay } = {}) {
    const q = params ? "?" + new URLSearchParams(Object.entries(params).filter(([, v]) => v != null && v !== "")).toString() : "";
    const headers = {
      "Accept": "application/json",
      ...(body && { "Content-Type": "application/json" }),
      ...(simulateDelay && { "x-simulate-delay": String(simulateDelay) }),
      ...(simulateError && { "x-simulate-error": "true" })
    };

    const res = await fetch(`${BASE}${url}${q}`, {
      method,
      headers,
      signal,
      body: body ? JSON.stringify(body) : null
    });

    if (!res.ok) {
      const err = new Error((await res.json().catch(() => ({}))).message || `HTTP Error ${res.status}`);
      err.status = res.status;
      err.client = "Fetch API";
      throw err;
    }

    return { data: await res.json(), status: res.status, client: "Fetch API" };
  },

  get(url, opt) { return this.request(url, { ...opt, method: "GET" }); },
  post(url, body, opt) { return this.request(url, { ...opt, method: "POST", body }); },
  put(url, body, opt) { return this.request(url, { ...opt, method: "PUT", body }); },
  delete(url, opt) { return this.request(url, { ...opt, method: "DELETE" }); }
};

/**
 * REST API Client implementation using Axios
 * Supports Full CRUD: GET, POST, PUT, and DELETE
 */
import axios from "axios";

const api = axios.create({ baseURL: "/api", timeout: 10000 });

const wrap = async (fn, endpoint, opt = {}, body = null) => {
  const { params, signal, simulateError, simulateDelay } = opt;
  const headers = {
    ...(simulateDelay && { "x-simulate-delay": String(simulateDelay) }),
    ...(simulateError && { "x-simulate-error": "true" })
  };

  try {
    const res = body !== null
      ? await fn(endpoint, body, { headers, signal })
      : await fn(endpoint, { params, headers, signal });
    return { data: res.data, status: res.status, client: "Axios Client" };
  } catch (err) {
    const e = new Error(err.response?.data?.message || err.message);
    e.status = err.response?.status || 500;
    e.client = "Axios Client";
    throw e;
  }
};

export const axiosClient = {
  get: (u, o) => wrap(api.get, u, o),
  post: (u, b, o) => wrap(api.post, u, o, b),
  put: (u, b, o) => wrap(api.put, u, o, b),
  delete: (u, o) => wrap(api.delete, u, o)
};

              title="Adds 2000ms delay to REST responses to test loading state"
            >
              <Clock size={14} />
              <span>{simulateDelay ? "Delay: 2s Active" : "Simulate 2s Delay"}</span>
            </button>

            <button

              title="Forces HTTP 500 responses to test error handling & retry"

import React, { useState, useEffect, useRef } from "react";
import {
  X,
  Calendar,
  Users,
  Clock,
  CheckCircle,
  AlertCircle,
  RefreshCw,
  Armchair,
  Sparkles,
  Phone,
  Mail,
  User,
  MessageSquare,
  ArrowRight,
  ArrowLeft,
  Check
} from "lucide-react";
import { getRestaurantAvailability, createReservation } from "../services/api.js";
import { useDispatch } from "react-redux";
import { updateDraft, confirmReservation as saveToStore } from "../store/reservationSlice.js";

export function ReservationModal({
  restaurant,
  isOpen,
  onClose,
  clientType,
  simulateDelay,
  simulateError,
  onRequestCompleted
}) {
  const dispatch = useDispatch();
  const [step, setStep] = useState(1); // 1: Select Table & Slot, 2: Guest Details, 3: Confirmation
  const [date, setDate] = useState(() => new Date().toISOString().split("T")[0]);
  const [partySize, setPartySize] = useState(2);

  // Availability state
  const [tables, setTables] = useState([]);
  const [availableSlots, setAvailableSlots] = useState([]);
  const [isLoadingAvailability, setIsLoadingAvailability] = useState(false);
  const [availabilityError, setAvailabilityError] = useState(null);

  // Selection
  const [selectedTable, setSelectedTable] = useState(null);
  const [selectedSlot, setSelectedSlot] = useState("");

  // Guest details form
  const [guest, setGuest] = useState({
    name: "",
    email: "",
    phone: "",
    specialRequests: ""
  });
  const [formErrors, setFormErrors] = useState({});

  // Submission state
  const [isSubmitting, setIsSubmitting] = useState(false);
  const [submitError, setSubmitError] = useState(null);
  const [confirmation, setConfirmation] = useState(null);

  const abortControllerRef = useRef(null);

  // Reset modal state on open/close
  useEffect(() => {
    if (isOpen) {
      setStep(1);
      setSelectedTable(null);
      setSelectedSlot("");
      setSubmitError(null);
      setConfirmation(null);
      fetchAvailabilityData();
    } else {
      if (abortControllerRef.current) {
        abortControllerRef.current.abort();
      }
    }
  }, [isOpen, restaurant?.id]);

  // Re-fetch availability when date or partySize changes
  useEffect(() => {
    if (isOpen && restaurant?.id) {
      setSelectedTable(null);
      fetchAvailabilityData();
    }
  }, [date, partySize]);

  async function fetchAvailabilityData() {
    if (!restaurant) return;

    if (abortControllerRef.current) {
      abortControllerRef.current.abort();
    }
    abortControllerRef.current = new AbortController();

    setIsLoadingAvailability(true);
    setAvailabilityError(null);

    try {
      const response = await getRestaurantAvailability(
        restaurant.id,
        { date, partySize },
        {
          signal: abortControllerRef.current.signal,
          simulateDelay: simulateDelay ? 2000 : null,
          simulateError,
          clientType
        }
      );

      if (onRequestCompleted) {
        onRequestCompleted(response.client);
      }

      const resData = response.data;
      setTables(resData.tables || []);
      setAvailableSlots(resData.availableSlots || []);

      // Auto-select first available table if appropriate
      const firstAvail = (resData.tables || []).find((t) => t.isAvailable);
      if (firstAvail) {
        setSelectedTable(firstAvail);
      }

      // Auto-select first slot
      if (resData.availableSlots && resData.availableSlots.length > 0) {
        setSelectedSlot(resData.availableSlots[0]);
      }

      dispatch(updateDraft({
        date,
        guests: partySize,
        restaurantName: restaurant.name,
        selectedTable: firstAvail || null,
        timeSlot: resData.availableSlots?.[0] || ""
      }));
    } catch (err) {
      if (err.isAborted || err.name === "AbortError") {
        return; // Normal cancellation
      }
      setAvailabilityError(err);
    } finally {
      setIsLoadingAvailability(false);
    }
  }

  function validateGuestForm() {
    const errors = {};
    if (!guest.name.trim()) errors.name = "Guest name is required.";
    if (!guest.email.trim()) {
      errors.email = "Email address is required.";
    } else if (!/\S+@\S+\.\S+/.test(guest.email)) {
      errors.email = "Please enter a valid email address.";
    }
    if (!guest.phone.trim()) {
      errors.phone = "Phone number is required.";
    } else if (guest.phone.replace(/\D/g, "").length < 7) {
      errors.phone = "Please provide a valid contact number.";
    }
    setFormErrors(errors);
    return Object.keys(errors).length === 0;
  }

  async function handleSubmitReservation(e) {
    e.preventDefault();
    if (!validateGuestForm()) return;

    setIsSubmitting(true);
    setSubmitError(null);

    const payload = {
      restaurantId: restaurant.id,
      restaurantName: restaurant.name,
      restaurantAddress: restaurant.address,
      date,
      timeSlot: selectedSlot,
      tableId: selectedTable?.id || "T1",
      tableNumber: selectedTable?.number || 1,
      tableType: selectedTable?.type || "Standard Table",
      partySize,
      customerName: guest.name,
      customerEmail: guest.email,
      customerPhone: guest.phone,
      specialRequests: guest.specialRequests,
      apiClientUsed: clientType
    };

    try {
      const response = await createReservation(payload, {
        simulateDelay: simulateDelay ? 2000 : null,
        simulateError,
        clientType
      });

      if (onRequestCompleted) {
        onRequestCompleted(response.client);
      }

      dispatch(saveToStore(response.data.data));
      setConfirmation(response.data.data);
      setStep(3); // Success step
    } catch (err) {
      setSubmitError(err);
    } finally {
      setIsSubmitting(false);
    }
  }

  if (!isOpen || !restaurant) return null;

  return (
    <div className="modal-backdrop" onClick={onClose}>
      <div className="modal-dialog" onClick={(e) => e.stopPropagation()}>
        {/* Modal Header */}
        <div className="modal-header">
          <div>
            <h2 className="modal-title">Table Reservation &bull; {restaurant.name}</h2>
            <div className="modal-subtitle">
              <span>{restaurant.cuisine} Cuisine</span> &bull; <span>{restaurant.location}</span> &bull;{" "}
              <span className="client-indicator">Active API Client: {clientType.toUpperCase()}</span>
            </div>
          </div>
          <button className="modal-close-btn" onClick={onClose} title="Close">
            <X size={20} />
          </button>
        </div>

        {/* Progress Stepper */}
        {step < 3 && (
          <div className="modal-stepper">
            <div className={`step-item ${step === 1 ? "current" : "completed"}`}>
              <span className="step-number">{step > 1 ? <Check size={14} /> : "1"}</span>
              <span className="step-label">Table &amp; Time Slot</span>
            </div>
            <div className="step-divider"></div>
            <div className={`step-item ${step === 2 ? "current" : ""}`}>
              <span className="step-number">2</span>
              <span className="step-label">Guest Details</span>
            </div>
          </div>
        )}

        {/* Modal Body */}
        <div className="modal-body">
          {/* STEP 1: Date, Party Size, Table, Time Slot Selection */}

              {/* Filter controls row */}

                    className="modal-select"

                    onChange={(e) => setPartySize(parseInt(e.target.value, 10))}
                  >
                    {[1, 2, 3, 4, 5, 6, 8, 10].map((num) => (
                      <option key={num} value={num}>
                        {num} {num === 1 ? "Guest" : "Guests"}
                      </option>

              {availabilityError && (

import React, { useState, useEffect } from "react";
import { useSelector, useDispatch } from "react-redux";

  X,

  Trash2,
  Edit3,
  CalendarX,
  RefreshCw,
  AlertCircle
} from "lucide-react";

export function MyBookingsModal({ isOpen, onClose, onOpenEdit, clientType }) {
  const dispatch = useDispatch();
  const reservations = useSelector((state) => state.reservation.confirmedReservations || []);

  const [isLoading, setIsLoading] = useState(false);
  const [error, setError] = useState(null);
  const [deletingId, setDeletingId] = useState(null);

  // Requirement e: Retrieve reservation history using GET
  const fetchHistory = async () => {
    setIsLoading(true);
    setError(null);
    try {
      const res = await getReservations({}, { clientType });
      dispatch(setConfirmedReservations(res.data.data || []));
    } catch (err) {
      setError(err.message || "Failed to load reservation history from REST API.");
    } finally {
      setIsLoading(false);
    }
  };

  useEffect(() => {
    if (isOpen) {
      fetchHistory();
    }
  }, [isOpen]);

  // Requirement d: Cancel a reservation using DELETE
  const handleDelete = async (id) => {
    setDeletingId(id);
    try {
      await deleteReservation(id, { clientType });
      dispatch(cancelReservation(id));
    } catch (err) {
      alert("Failed to cancel reservation via DELETE: " + err.message);
    } finally {
      setDeletingId(null);
    }
  };

  if (!isOpen) return null;

  return (
    <div className="modal-backdrop" onClick={onClose}>
      <div className="modal-dialog" onClick={(e) => e.stopPropagation()} style={{ maxWidth: 680 }}>
        <div className="modal-header">
          <div>
            <h3 className="modal-title">My Table Reservations</h3>
            <div className="modal-subtitle">
              <span>REST CRUD: GET history &bull; PUT edit &bull; DELETE cancel</span> &bull; <strong style={{ color: "var(--primary)" }}>{clientType.toUpperCase()}</strong>
            </div>
          </div>
          <button className="modal-close-btn" onClick={onClose} title="Close">
            <X size={20} />
          </button>
        </div>

        <div className="modal-body" style={{ maxHeight: "65vh" }}>
          {/* Error handling */}
          {error && (
            <div className="modal-alert modal-error" style={{ marginBottom: "1rem" }}>
              <AlertCircle size={16} />
              <span>{error}</span>
              <button className="btn-alert-action" onClick={fetchHistory}>
                <RefreshCw size={13} /> Retry
              </button>
            </div>
          )}

          {/* Loading state */}
          {isLoading ? (
            <div className="modal-loading-state" style={{ padding: "3rem 1rem" }}>
              <div className="spinner"></div>
              <p>Fetching reservation history via GET /api/reservations ({clientType.toUpperCase()})...</p>
            </div>
          ) : reservations.length === 0 ? (
            /* Requirement f: Display appropriate empty state */
            <div className="empty-state" style={{ margin: "1rem auto", padding: "2.5rem 1rem" }}>
              <CalendarX size={44} className="empty-icon" style={{ color: "#94a3b8" }} />
              <h4 className="empty-title">No Reservations Found</h4>
              <p className="empty-text">
                No active bookings in the REST database. Reserve a table to begin!
              </p>

import React, { useState, useEffect } from "react";

import { updateReservation } from "../services/api.js";
import { X, Calendar, Clock, Users, MessageSquare, Phone, Save, AlertCircle } from "lucide-react";

export function EditReservationModal({ reservation, isOpen, onClose, clientType }) {
  const dispatch = useDispatch();

  const [date, setDate] = useState("");
  const [timeSlot, setTimeSlot] = useState("");
  const [partySize, setPartySize] = useState(2);
  const [customerPhone, setCustomerPhone] = useState("");
  const [specialRequests, setSpecialRequests] = useState("");

  const [isUpdating, setIsUpdating] = useState(false);
  const [updateError, setUpdateError] = useState(null);

  useEffect(() => {
    if (reservation) {
      setDate(reservation.date || "");
      setTimeSlot(reservation.timeSlot || "7:00 PM");
      setPartySize(reservation.partySize || 2);
      setCustomerPhone(reservation.customerPhone || "");
      setSpecialRequests(reservation.specialRequests || "");
      setUpdateError(null);
    }
  }, [reservation]);

  if (!isOpen || !reservation) return null;

  const handleUpdate = async (e) => {
    e.preventDefault();
    setIsUpdating(true);
    setUpdateError(null);

    const payload = {
      date,
      timeSlot,
      partySize: parseInt(partySize, 10),
      customerPhone,
      specialRequests
    };

    try {
      // HTTP PUT /api/reservations/:id (Requirement c)
      const res = await updateReservation(reservation.id, payload, { clientType });
      const updated = res.data.data;
      dispatch(updateReservationInState(updated));

      setUpdateError(err.message || "Failed to update reservation via PUT request.");
    } finally {
      setIsUpdating(false);
    }

import React, { useState, useEffect, useRef, useCallback } from "react";
import { useDispatch } from "react-redux";
import { setSelectedRestaurant as setStoreSelectedRestaurant } from "./store/reservationSlice.js";
import { Navbar } from "./components/Navbar.jsx";
import { SearchFilterBar } from "./components/SearchFilterBar.jsx";
import { RestaurantCard } from "./components/RestaurantCard.jsx";
import { LoadingSkeleton } from "./components/LoadingSkeleton.jsx";
import { ErrorMessage } from "./components/ErrorMessage.jsx";
import { ReservationModal } from "./components/ReservationModal.jsx";
import { MyBookingsModal } from "./components/MyBookingsModal.jsx";
import { EditReservationModal } from "./components/EditReservationModal.jsx";
import { MiddlewareMonitor } from "./components/MiddlewareMonitor.jsx";
import { getRestaurants } from "./services/api.js";
import { RESTAURANTS_DATA } from "./data/mockRestaurants.js";
import { Sparkles, Frown } from "lucide-react";

export function App() {
  const dispatch = useDispatch();

  // REST API Client selection ('axios' | 'fetch')
  const [clientType, setClientType] = useState("axios");

  // Simulation toggles for testing loading & error handling
  const [simulateDelay, setSimulateDelay] = useState(false);
  const [simulateError, setSimulateError] = useState(false);

  // Request counter & tracking
  const [requestStats, setRequestStats] = useState({
    count: 0,
    lastClient: "Axios Client"
  });

  // Filter state
  const [filters, setFilters] = useState({
    search: "",
    cuisine: "All Cuisines",
    location: "All Locations",
    minRating: 0,
    priceLevel: "all"
  });

  // Restaurant data state
  const [restaurants, setRestaurants] = useState([]);
  const [isLoading, setIsLoading] = useState(true);
  const [error, setError] = useState(null);

  // Modal dialog states
  const [selectedRestaurant, setSelectedRestaurant] = useState(null);
  const [isModalOpen, setIsModalOpen] = useState(false);
  const [isBookingsOpen, setIsBookingsOpen] = useState(false);
  const [editingReservation, setEditingReservation] = useState(null);

  // AbortController reference to cancel in-flight requests on rapid filter changes
  const abortControllerRef = useRef(null);

  // Fetch restaurants from REST API (GET /api/restaurants)
  const fetchRestaurantsData = useCallback(async () => {
    if (abortControllerRef.current) {
      abortControllerRef.current.abort();
    }
    abortControllerRef.current = new AbortController();

    setIsLoading(true);
    setError(null);

    try {
      const response = await getRestaurants(
        {
          search: filters.search,
          cuisine: filters.cuisine,
          location: filters.location,
          minRating: filters.minRating > 0 ? filters.minRating : undefined,
          priceLevel: filters.priceLevel !== "all" ? filters.priceLevel : undefined
        },
        {
          signal: abortControllerRef.current.signal,
          simulateDelay: simulateDelay ? 2000 : null,
          simulateError,
          clientType
        }
      );

      setRestaurants(response.data.data || []);
      setRequestStats((prev) => ({
        count: prev.count + 1,
        lastClient: response.client
      }));
    } catch (err) {
      if (err.isAborted || err.name === "AbortError") {
        return;
      }
      setError(err);
    } finally {
      setIsLoading(false);
    }
  }, [filters, simulateDelay, simulateError, clientType]);

  useEffect(() => {
    fetchRestaurantsData();

    return () => {
      if (abortControllerRef.current) {
        abortControllerRef.current.abort();
      }
    };
  }, [fetchRestaurantsData]);

  const handleFilterChange = (key, value) => {
    setFilters((prev) => ({
      ...prev,
      [key]: value
    }));
  };

  const handleResetFilters = () => {
    setFilters({
      search: "",
      cuisine: "All Cuisines",
      location: "All Locations",
      minRating: 0,
      priceLevel: "all"
    });
  };

  // Open booking modal (POST)
  const handleOpenBooking = (restaurant) => {
    dispatch(setStoreSelectedRestaurant(restaurant));
    setSelectedRestaurant(restaurant);
    setIsModalOpen(true);
  };

  const handleCloseBooking = () => {
    setIsModalOpen(false);
    setSelectedRestaurant(null);
  };

  const handleRequestCompleted = (client) => {
    setRequestStats((prev) => ({
      count: prev.count + 1,
      lastClient: client
    }));
  };

  return (
    <div className="app-container">
      {/* Top Navigation */}
      <Navbar
        clientType={clientType}
        onClientTypeChange={(type) => setClientType(type)}
        simulateDelay={simulateDelay}
        onToggleSimulateDelay={() => setSimulateDelay((prev) => !prev)}
        simulateError={simulateError}
        onToggleSimulateError={() => setSimulateError((prev) => !prev)}
        requestStats={requestStats}
        onOpenBookings={() => setIsBookingsOpen(true)}
      />

      <main className="main-content">
        {/* Hero Section */}
        <section className="hero-section">
          <div className="hero-badge">
            <Sparkles size={14} />
            <span>Exp 3: Full Restaurant Reservation CRUD Operations (GET, POST, PUT, DELETE)</span>

          <h1 className="hero-title">
            Reserve &amp; Manage Your Dining Reservations
          </h1>
          <p className="hero-subtitle">
            Complete RESTful lifecycle powered by <strong>Axios &amp; Fetch</strong>: retrieve listings (<strong>GET</strong>),
            create reservations (<strong>POST</strong>), modify bookings (<strong>PUT</strong>), and cancel reservations (<strong>DELETE</strong>).
          </p>
        </section>

        {/* Search & Filter Bar */}
        <SearchFilterBar
          filters={filters}
          onFilterChange={handleFilterChange}
          onResetFilters={handleResetFilters}
          totalResults={restaurants.length}
          totalAvailable={RESTAURANTS_DATA.length}
          isSearching={isLoading}
        />

        {/* Results Section */}
        <section className="results-section">
          {error && (
            <ErrorMessage
              error={error}
              onRetry={fetchRestaurantsData}
              title="Error Loading Restaurants"
            />
          )}

          {isLoading ? (
            <LoadingSkeleton count={6} />
          ) : !error && restaurants.length === 0 ? (
            <div className="empty-state">
              <Frown size={48} className="empty-icon" />
              <h3 className="empty-title">No Matching Restaurants Found</h3>
              <p className="empty-text">
                We couldn't find any restaurants matching your current search or filter combination.
              </p>
              <button className="btn-primary" onClick={handleResetFilters}>
                Clear All Filters
              </button>
            </div>
          ) : (
            <div className="restaurant-grid">
              {restaurants.map((restaurant) => (
                <RestaurantCard
                  key={restaurant.id}
                  restaurant={restaurant}
                  onSelectForBooking={handleOpenBooking}
                />
              ))}
            </div>
          )}
        </section>
      </main>

      {/* Reservation Booking Modal (POST) */}
      <ReservationModal
        restaurant={selectedRestaurant}
        isOpen={isModalOpen}
        onClose={handleCloseBooking}
        clientType={clientType}
        simulateDelay={simulateDelay}
        simulateError={simulateError}
        onRequestCompleted={handleRequestCompleted}
      />

      {/* My Bookings History Modal (GET & DELETE) */}
      <MyBookingsModal
        isOpen={isBookingsOpen}
        onClose={() => setIsBookingsOpen(false)}
        onOpenEdit={(res) => setEditingReservation(res)}
        clientType={clientType}
      />

      {/* Modify Reservation Modal (PUT) */}
      <EditReservationModal
        reservation={editingReservation}
        isOpen={!!editingReservation}
        onClose={() => setEditingReservation(null)}
        clientType={clientType}
      />

      {/* Real-time Redux Action Monitoring Middleware Panel */}
      <MiddlewareMonitor />

      {/* Footer */}
      <footer className="app-footer">
        <div className="footer-content">
          <p>
            <strong>Scenario 5 &bull; Experiment 3:</strong> Restaurant Reservation CRUD Operations
          </p>
          <p className="footer-sub">
            Demonstrates GET (History/Tables), POST (Create), PUT (Modify), and DELETE (Cancel) with Axios &amp; Fetch API.
          </p>
        </div>

function restApiPlugin() {
  return {
    name: "mock-rest-api",
    configureServer(server) {
      server.middlewares.use((req, res, next) => {
        const parsedUrl = new URL(req.url, `http://${req.headers.host || "localhost"}`);
        const pathname = parsedUrl.pathname;

        if (!pathname.startsWith("/api/")) {
          return next();
        }

        const simulateDelay = req.headers["x-simulate-delay"] || parsedUrl.searchParams.get("delay");
        const simulateError = req.headers["x-simulate-error"] === "true" || parsedUrl.searchParams.get("error") === "true";
        const delayMs = simulateDelay ? parseInt(simulateDelay, 10) : 300;

        setTimeout(() => {
          res.setHeader("Content-Type", "application/json");

          if (simulateError) {
            res.statusCode = 500;
            res.end(JSON.stringify({
              success: false,
              status: 500,
              message: "Simulated Server Error: Failed to process REST CRUD operation."
            }));
            return;
          }

          // 1. GET /api/restaurants (Retrieve restaurants)
          if (pathname === "/api/restaurants" && req.method === "GET") {
            const search = (parsedUrl.searchParams.get("search") || "").trim().toLowerCase();
            const cuisine = parsedUrl.searchParams.get("cuisine") || "All Cuisines";
            const location = parsedUrl.searchParams.get("location") || "All Locations";
            const minRating = parseFloat(parsedUrl.searchParams.get("minRating") || "0");
            const priceLevel = parsedUrl.searchParams.get("priceLevel");

            let filtered = [...RESTAURANTS_DATA];
            if (search) {
              filtered = filtered.filter(r =>
                r.name.toLowerCase().includes(search) ||
                r.cuisine.toLowerCase().includes(search) ||
                r.location.toLowerCase().includes(search) ||
                r.description.toLowerCase().includes(search) ||
                r.tags.some(t => t.toLowerCase().includes(search))
              );
            }
            if (cuisine && cuisine !== "All Cuisines") {
              filtered = filtered.filter(r => r.cuisine.toLowerCase() === cuisine.toLowerCase());
            }
            if (location && location !== "All Locations") {
              filtered = filtered.filter(r => r.location.toLowerCase() === location.toLowerCase());
            }
            if (minRating > 0) {
              filtered = filtered.filter(r => r.rating >= minRating);
            }
            if (priceLevel && priceLevel !== "all") {
              filtered = filtered.filter(r => r.priceLevel === parseInt(priceLevel, 10));
            }

            res.statusCode = 200;
            res.end(JSON.stringify({
              success: true,
              count: filtered.length,
              timestamp: new Date().toISOString(),
              data: filtered
            }));
            return;
          }

          // 2. GET /api/restaurants/:id/availability (Retrieve tables and time slots)
          const availabilityMatch = pathname.match(/^\/api\/restaurants\/(\d+)\/availability$/);
          if (availabilityMatch && req.method === "GET") {
            const restaurantId = parseInt(availabilityMatch[1], 10);
            const restaurant = RESTAURANTS_DATA.find(r => r.id === restaurantId);

            if (!restaurant) {
              res.statusCode = 404;
              res.end(JSON.stringify({ success: false, status: 404, message: `Restaurant ID ${restaurantId} not found.` }));
              return;
            }

            const date = parsedUrl.searchParams.get("date") || new Date().toISOString().split("T")[0];
            const partySize = parseInt(parsedUrl.searchParams.get("partySize") || "2", 10);
            const dateSeed = date.split("-").reduce((acc, part) => acc + parseInt(part, 10), 0);

            const tablesWithDynamicStatus = restaurant.tables.map(table => {
              const tableNumeric = parseInt(table.number, 10) || 1;
              const isAvailable = (tableNumeric + dateSeed + partySize) % 5 !== 0 && table.capacity >= partySize;
              return {
                ...table,
                isAvailable,
                capacitySufficient: table.capacity >= partySize
              };
            });

            const availableSlots = restaurant.timeSlots.filter((_, idx) => (idx + dateSeed) % 4 !== 0);

            res.statusCode = 200;
            res.end(JSON.stringify({
              success: true,
              restaurantId,
              restaurantName: restaurant.name,
              date,
              partySize,
              tables: tablesWithDynamicStatus,
              availableSlots: availableSlots.length > 0 ? availableSlots : restaurant.timeSlots.slice(0, 3)
            }));
            return;
          }

          // 3. GET /api/reservations (Retrieve reservation history)
          if (pathname === "/api/reservations" && req.method === "GET") {
            res.statusCode = 200;
            res.end(JSON.stringify({
              success: true,
              count: reservationsDB.length,
              timestamp: new Date().toISOString(),
              data: reservationsDB
            }));
            return;
          }

          // 4. POST /api/reservations (Create a reservation)
          if (pathname === "/api/reservations" && req.method === "POST") {
            let body = "";
            req.on("data", chunk => { body += chunk.toString(); });
            req.on("end", () => {
              try {
                const parsedBody = body ? JSON.parse(body) : {};
                const confirmationCode = "RES-" + Math.floor(100000 + Math.random() * 900000);
                const reservation = {
                  id: "res_" + Date.now(),
                  confirmationCode,
                  createdAt: new Date().toISOString(),
                  ...parsedBody
                };

                reservationsDB.unshift(reservation);

                res.statusCode = 201;
                res.end(JSON.stringify({
                  success: true,
                  message: "Reservation created successfully (POST 201 Created)",
                  data: reservation
                }));
              } catch (err) {
                res.statusCode = 400;
                res.end(JSON.stringify({ success: false, status: 400, message: "Invalid JSON payload: " + err.message }));
              }
            });
            return;
          }

          // 5. PUT or PATCH /api/reservations/:id (Modify reservation details)
          const reservationItemMatch = pathname.match(/^\/api\/reservations\/([^/]+)$/);
          if (reservationItemMatch && (req.method === "PUT" || req.method === "PATCH")) {
            const reservationId = reservationItemMatch[1];
            let body = "";
            req.on("data", chunk => { body += chunk.toString(); });
            req.on("end", () => {
              try {
                const updates = body ? JSON.parse(body) : {};
                const index = reservationsDB.findIndex(r => r.id === reservationId);

                if (index === -1) {
                  res.statusCode = 404;
                  res.end(JSON.stringify({ success: false, status: 404, message: `Reservation ${reservationId} not found.` }));
                  return;
                }

                reservationsDB[index] = {
                  ...reservationsDB[index],
                  ...updates,
                  updatedAt: new Date().toISOString()
                };

                res.statusCode = 200;
                res.end(JSON.stringify({
                  success: true,
                  message: `Reservation ${reservationId} updated successfully (PUT 200 OK)`,
                  data: reservationsDB[index]
                }));
              } catch (err) {
                res.statusCode = 400;
                res.end(JSON.stringify({ success: false, status: 400, message: "Invalid update payload: " + err.message }));
              }
            });
            return;
          }

          // 6. DELETE /api/reservations/:id (Cancel a reservation)
          if (reservationItemMatch && req.method === "DELETE") {
            const reservationId = reservationItemMatch[1];
            const index = reservationsDB.findIndex(r => r.id === reservationId);

            if (index === -1) {
              res.statusCode = 404;
              res.end(JSON.stringify({ success: false, status: 404, message: `Reservation ${reservationId} not found.` }));
              return;
            }

            const removed = reservationsDB.splice(index, 1)[0];
            res.statusCode = 200;
            res.end(JSON.stringify({
              success: true,
              message: `Reservation ${reservationId} cancelled successfully (DELETE 200 OK)`,
              data: removed
            }));
            return;
          }

          // Fallback 404
          res.statusCode = 404;
          res.end(JSON.stringify({ success: false, status: 404, message: `Endpoint ${pathname} not found.` }));
        }, delayMs);
      });
    }
  };
}

export default defineConfig({
  plugins: [react(), restApiPlugin()],
  server: {
    port: 5173,
    host: true
  }
});
